      square=FNmousepos

      CASE TRUE OF

            IF xpos=1 THEN S%=12
            IF xpos=xsize THEN S%=14
            PROCstore(xpos,1,S%)
          NEXT

          FOR ypos=1 TO ysize
            FOR xpos=1 TO xsize
              IF ypos=1 THEN
                S%=17
                IF xpos=1 THEN S%=7
                IF xpos=xsize THEN S%=6
              ENDIF
              IF ypos=ysize THEN
                S%=15
                IF xpos=1 THEN S%=8
                IF xpos=xsize THEN S%=5
              ENDIF
              IF ypos<>ysize AND ypos<>1 THEN
                S%=4
                IF xpos=1 THEN S%=16
                IF xpos=xsize THEN S%=18
              ENDIF
              PROCstore(xpos,ypos,S%)
            NEXT
          NEXT

      PROCbutton_up

        WHEN 37:lefttop=manpos+dir-41:IF dir=1  THEN direction=-1
        WHEN 38:lefttop=manpos+dir-20:IF dir=20 THEN direction=-1

        ENDIF

      CASE dir OF
        WHEN -1,1:

              Filed%(Filpntr)=segment+dir:Filpntr+=1
              sprite%(segment+dir) = sprite%(segment)
              block%(segment+dir)  = block%(segment)
              segment-=dir
            NEXT
          NEXT
          segment=manpos+dir+20*(ypos-1)

      REPEAT : OFF

        ON : K% = GET
        CASE K% OF
          WHEN 8,127: IF C% A$ = LEFT$(A$,C%-1) + MID$(A$,C%+1) : C% -= 1
          WHEN 13:
          WHEN 136: IF C% C% -= 1
          WHEN 137: IF C% < LEN(A$) C% += 1
