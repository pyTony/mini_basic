      DEF PROCwaterfall(X%, RETURN Y%, font$)
      LOCAL H%
      Y% -= 10
      FOR H% = 10 TO 22 STEP 3
        Y% -= H%
        PROC_PLfont(font$,H%,0) : REM colour=0=black
        PROC_PLprint(X%, Y%, "abcdefghijklmnopqrstuvwxyz £€\1234567890 ")
      NEXT
      Y% -= 24
      ENDPROC
